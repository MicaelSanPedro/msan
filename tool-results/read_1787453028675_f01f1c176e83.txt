    47→function getStored(key: string): string | null {
    48→  if (typeof window === "undefined") return null;
    49→  try { return localStorage.getItem(key); } catch { return null; }
    50→}
    51→function setStored(key: string, val: string) {
    52→  try { localStorage.setItem(key, val); } catch { /* ignore */ }
    53→}
    54→function removeStored(key: string) {
    55→  try { localStorage.removeItem(key); } catch { /* ignore */ }
    56→}
    57→
    58→function getStorageSize(): string {
    59→  if (typeof window === "undefined") return "0 B";
    60→  try {
    61→    let total = 0;
    62→    for (const key of Object.keys(localStorage)) {
    63→      total += (localStorage.getItem(key) || "").length;
    64→    }
    65→    if (total < 1024) return `${total} B`;
    66→    return `${(total / 1024).toFixed(1)} KB`;
    67→  } catch { return "0 B"; }
    68→}
    69→function getVisitCount(): number {
    70→  try {
    71→    const stored = localStorage.getItem("techmate_visits");
    72→    if (stored) {
    73→      const count = parseInt(stored, 10) + 1;
    74→      localStorage.setItem("techmate_visits", String(count));
    75→      return count;
    76→    }
    77→    localStorage.setItem("techmate_visits", "1");
    78→    return 1;
    79→  }
    80→  catch { return 1; }
    81→}
    82→function getFirstVisit(): string {
    83→  try {
    84→    const stored = localStorage.getItem("techmate_first_visit");
    85→    if (stored) return stored;
    86→    const now = new Date();
    87→    const formatted = `${now.getDate().toString().padStart(2, "0")}/${(now.getMonth() + 1).toString().padStart(2, "0")}/${now.getFullYear()}`;
    88→    localStorage.setItem("techmate_first_visit", formatted);
    89→    return formatted;
    90→  }
    91→  catch { return "Hoje"; }
    92→}

   192→/* Segmented control for selecting from options */
   193→function SegmentedControl<T extends string>({
   194→  options,
   195→  value,
   196→  onChange,
   197→  renderLabel,
   198→}: {
   199→  options: readonly { label: string; value: T; desc?: string }[];
   200→  value: T;
   201→  onChange: (val: T) => void;
   202→  renderLabel?: (opt: { label: string; value: T; desc?: string }) => string;
   203→}) {
   204→  return (
   205→    <div className="flex flex-wrap gap-1.5">
   206→      {options.map((opt) => {
   207→        const isActive = value === opt.value;
   208→        return (
   209→          <button
   210→            key={opt.value}
   211→            type="button"
   212→            onClick={() => onChange(opt.value)}
   213→            className={`px-3 py-1.5 rounded-lg text-xs font-medium transition-all duration-200 cursor-pointer
   214→                       ${isActive
   215→                         ? "bg-emerald-500/20 border border-emerald-500/30 text-emerald-300 shadow-[0_0_12px_rgba(52,211,153,0.15)]"
   216→                         : "bg-white/[0.03] border border-white/30 text-white/40 hover:bg-white/[0.06] hover:text-white/60"}`}
   217→          >
   218→            {renderLabel ? renderLabel(opt) : opt.label}
   219→            {isActive && <Check className="w-3 h-3 inline ml-1" />}
   220→          </button>
   221→        );
   222→      })}
   223→    </div>
   224→  );
   225→}
