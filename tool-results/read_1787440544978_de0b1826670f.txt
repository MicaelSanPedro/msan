   214→/* Segmented control for selecting from options */
   215→function SegmentedControl<T extends string>({
   216→  options,
   217→  value,
   218→  onChange,
   219→  renderLabel,
   220→}: {
   221→  options: readonly { label: string; value: T; desc?: string }[];
   222→  value: T;
   223→  onChange: (val: T) => void;
   224→  renderLabel?: (opt: { label: string; value: T; desc?: string }) => string;
   225→}) {
   226→  return (
   227→    <div className="flex flex-wrap gap-1.5">
   228→      {options.map((opt) => {
   229→        const isActive = value === opt.value;
   230→        return (
   231→          <button
   232→            key={opt.value}
   233→            type="button"
   234→            onClick={() => onChange(opt.value)}
   235→            className={`px-3 py-1.5 rounded-lg text-xs font-medium transition-all duration-200 cursor-pointer
   236→                       ${isActive
   237→                         ? "bg-emerald-500/20 border border-emerald-500/30 text-emerald-300 shadow-[0_0_12px_rgba(52,211,153,0.15)]"
   238→                         : "bg-white/[0.03] border border-white/30 text-white/40 hover:bg-white/[0.06] hover:text-white/60"}`}
   239→          >
   240→            {renderLabel ? renderLabel(opt) : opt.label}
   241→            {isActive && <Check className="w-3 h-3 inline ml-1" />}
   242→          </button>
   243→        );
   244→      })}
   245→    </div>
   246→  );
   247→}
