   137→/* =============================================
   138→   SEARCH ENGINE — Fuzzy matching
   139→   ============================================= */
   140→
   141→function normalize(str: string): string {
   142→  return str
   143→    .toLowerCase()
   144→    .normalize("NFD")
   145→    .replace(/[\u0300-\u036f]/g, "")
   146→    .replace(/[^a-z0-9\s]/g, "")
   147→    .trim();
   148→}
   149→
   150→function score(query: string, item: SearchResult): number {
   151→  const q = normalize(query);
   152→  const qWords = q.split(/\s+/).filter(Boolean);
   153→
   154→  const titleNorm = normalize(item.title);
   155→  const descNorm = normalize(item.description);
   156→  const catNorm = normalize(item.category);
   157→  const secNorm = normalize(item.section);
   158→  const combined = `${titleNorm} ${descNorm} ${catNorm} ${secNorm}`;
   159→
   160→  let score = 0;
   161→
   162→  // Exact title match
   163→  if (titleNorm === q) score += 100;
   164→  // Title starts with query
   165→  else if (titleNorm.startsWith(q)) score += 80;
   166→  // All query words in title
   167→  else if (qWords.every(w => titleNorm.includes(w))) score += 60;
   168→  // Query words in title partially
   169→  else if (qWords.some(w => titleNorm.includes(w))) score += 30;
   170→
   171→  // All words in combined text
   172→  if (qWords.every(w => combined.includes(w))) score += 25;
   173→  // Some words in combined text
   174→  else if (qWords.some(w => combined.includes(w))) score += 10;
   175→
   176→  // Sequential character matching (fuzzy)
   177→  let ci = 0;
   178→  for (let qi = 0; qi < q.length && ci < titleNorm.length; qi++) {
   179→    if (q[qi] === titleNorm[ci] || ci < titleNorm.length) {
   180→      while (ci < titleNorm.length && titleNorm[ci] !== q[qi]) ci++;
   181→      if (ci < titleNorm.length && titleNorm[ci] === q[qi]) score += 3;
   182→      ci++;
   183→    }
   184→  }
   185→
   186→  // Category/section bonus
   187→  if (catNorm.includes(q)) score += 15;
   188→  if (secNorm.includes(q)) score += 15;
   189→
   190→  return score;
   191→}
   192→
   193→function search(query: string): SearchResult[] {
   194→  if (!query || query.trim().length < 2) return [];
   195→  const q = normalize(query);
   196→  if (q.length < 2) return [];
   197→
   198→  const results = searchIndex
   199→    .map(item => ({ item, s: score(q, item) }))
   200→    .filter(r => r.s > 0)
   201→    .sort((a, b) => b.s - a.s)
   202→    .slice(0, 8)
   203→    .map(r => r.item);
   204→
   205→  return results;
   206→}

   295→          onClick={() => setOpen(false)}
   296→        >

   299→
