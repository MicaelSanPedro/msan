    46→function getStored(key: string): string | null {
    47→  if (typeof window === "undefined") return null;
    48→  try { return localStorage.getItem(key); } catch { return null; }
    49→}
    50→function setStored(key: string, val: string) {
    51→  try { localStorage.setItem(key, val); } catch { /* ignore */ }
    52→}
    53→function removeStored(key: string) {
    54→  try { localStorage.removeItem(key); } catch { /* ignore */ }
    55→}
    56→
    57→function getStorageSize(): string {
    58→  if (typeof window === "undefined") return "0 B";
    59→  try {
    60→    let total = 0;
    61→    for (const key of Object.keys(localStorage)) {
    62→      total += (localStorage.getItem(key) || "").length;
    63→    }
    64→    if (total < 1024) return `${total} B`;
    65→    return `${(total / 1024).toFixed(1)} KB`;
    66→  } catch { return "0 B"; }
    67→}
    68→function getVisitCount(): number {
    69→  try {
    70→    const stored = localStorage.getItem("techmate_visits");
    71→    if (stored) {
    72→      const count = parseInt(stored, 10) + 1;
    73→      localStorage.setItem("techmate_visits", String(count));
    74→      return count;
    75→    }
    76→    localStorage.setItem("techmate_visits", "1");
    77→    return 1;
    78→  }
    79→  catch { return 1; }
    80→}
    81→function getFirstVisit(): string {
    82→  try {
    83→    const stored = localStorage.getItem("techmate_first_visit");
    84→    if (stored) return stored;
    85→    const now = new Date();
    86→    const formatted = `${now.getDate().toString().padStart(2, "0")}/${(now.getMonth() + 1).toString().padStart(2, "0")}/${now.getFullYear()}`;
    87→    localStorage.setItem("techmate_first_visit", formatted);
    88→    return formatted;
    89→  }
    90→  catch { return "Hoje"; }
    91→}

   215→/* Segmented control for selecting from options */
   216→function SegmentedControl<T extends string>({
   217→  options,
   218→  value,
   219→  onChange,
   220→  renderLabel,
   221→}: {
   222→  options: readonly { label: string; value: T; desc?: string }[];
   223→  value: T;
   224→  onChange: (val: T) => void;
   225→  renderLabel?: (opt: { label: string; value: T; desc?: string }) => string;
   226→}) {
   227→  return (
   228→    <div className="flex flex-wrap gap-1.5">
   229→      {options.map((opt) => {
   230→        const isActive = value === opt.value;
   231→        return (
   232→          <button
   233→            key={opt.value}
   234→            type="button"
   235→            onClick={() => onChange(opt.value)}
   236→            className={`px-3 py-1.5 rounded-lg text-xs font-medium transition-all duration-200 cursor-pointer
   237→                       ${isActive
   238→                         ? "bg-amber-500/20 border border-amber-500/30 text-amber-300 shadow-[0_0_12px_rgba(249,189,24,0.15)]"
   239→                         : "bg-white/[0.03] border border-white/[0.08] text-white/40 hover:bg-white/[0.06] hover:text-white/60"}`}
   240→          >
   241→            {renderLabel ? renderLabel(opt) : opt.label}
   242→            {isActive && <Check className="w-3 h-3 inline ml-1" />}
   243→          </button>
   244→        );
   245→      })}
   246→    </div>
   247→  );
   248→}

   299→  const [storageSize, setStorageSize] = useState("0 B");
   300→  const [visitCount, setVisitCount] = useState(1);
   301→  const [firstVisit, setFirstVisit] = useState("Hoje");
   302→  const [savedMsg, setSavedMsg] = useState("");
   303→
   304→  const hydrate = useCallback(() => {
   305→    setMounted(true);
   306→
   307→    const t = getStored(KEYS.theme);
   308→    if (t === "light") setTheme("light");
   309→
   310→    const fs = getStored(KEYS.fontSize);
   311→    if (fs) setFontSize(fs);
   312→
   313→    const rm = getStored(KEYS.reducedMotion);
   314→    if (rm === "true") setReducedMotion(true);
   315→
   316→    const cm = getStored(KEYS.compactMode);
   317→    if (cm === "true") setCompactMode(true);
   318→
   319→    const ac = getStored(KEYS.accentColor);
   320→    if (ac) setAccentColor(ac);
   321→
   322→    setStorageSize(getStorageSize());
   323→    setVisitCount(getVisitCount());
   324→    setFirstVisit(getFirstVisit());
   325→  }, []);
   326→
   327→  useEffect(() => { hydrate(); }, [hydrate]);
   328→
   329→  /* ── Apply preferences to document ── */
   330→  const applyFontSize = useCallback((size: string) => {
   331→    const root = document.documentElement;
   332→    const preset = FONT_SIZES.find((f) => f.value === size);
   333→    if (preset) {
   334→      const scale = preset.scale / 16;
   335→      root.style.setProperty("--user-font-scale", String(scale));
   336→      root.style.fontSize = `${preset.scale}px`;
   337→    } else {
   338→      root.style.removeProperty("--user-font-scale");
   339→      root.style.fontSize = "";
   340→    }
   341→  }, []);
   342→
   343→  const applyReducedMotion = useCallback((enabled: boolean) => {
   344→    const root = document.documentElement;
   345→    if (enabled) {
   346→      root.classList.add("reduced-motion");
   347→    } else {
   348→      root.classList.remove("reduced-motion");
   349→    }
   350→  }, []);
   351→
   352→  const applyCompactMode = useCallback((enabled: boolean) => {
   353→    const root = document.documentElement;
   354→    if (enabled) {
   355→      root.classList.add("compact-mode");
   356→    } else {
   357→      root.classList.remove("compact-mode");
   358→    }
   359→  }, []);
   360→
   361→  const applyAccentColor = useCallback((color: string) => {
   362→    const preset = ACCENT_COLORS.find((c) => c.value === color);
   363→    if (!preset) return;
   364→    const root = document.documentElement;
   365→    root.style.setProperty("--accent", preset.primary);
   366→    root.style.setProperty("--accent-glow", preset.glow);
   367→    root.setAttribute("data-accent", color);
   368→  }, []);
   369→
   370→  /* Apply on mount */
   371→  useEffect(() => {
   372→    if (!mounted) return;
   373→    applyFontSize(fontSize);
   374→    applyReducedMotion(reducedMotion);
   375→    applyCompactMode(compactMode);
   376→    applyAccentColor(accentColor);
   377→  }, [mounted, fontSize, reducedMotion, compactMode, accentColor, applyFontSize, applyReducedMotion, applyCompactMode, applyAccentColor]);

   379→  const showSaved = useCallback(() => {

   384→  /* ── Handlers ── */

   392→      if (m) m.content = "#f8f7f5";

   394→      document.documentElement.classList.add("dark");
