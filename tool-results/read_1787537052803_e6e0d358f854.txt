   142→   241→   241→                this._observer = new MutationObserver(() => {
   143→   242→   242→                        // Structural mutation: painting order may have shifted,
   144→   243→   243→                        // every glass needs to re-render.

   149→   248→   248→
   150→   249→   249→                this._glassSubtreeObserver = new MutationObserver((mutations) => {
   151→   250→   250→                        for (const mutation of mutations) {
   152→   251→   251→                                const owner = this._closestGlassAncestor(mutation.target);
   153→   252→   252→                                if (mutation.type === 'attributes' && mutation.attributeName === 'data-config') {
   154→   253→   253→                                        // Config change → just re-shade this glass + any
   155→   254→   254→                                        // glass that depends on its rendered output.
   156→   255→   255→                                        if (owner) this._markGlassAndDependents(owner);
   157→   256→   256→                                        continue;
   158→   257→   257→                                }

   160→   259→   259→                                // to be re-captured AND its shader re-run (since the
   161→   260→   260→                                // content image is what we composite on top of the
   162→   261→   261→                                // shader output).
   163→   262→   262→                                if (owner) {
   164→   263→   263→                                        this._glassContentDirty.add(owner);
   165→   264→   264→                                        this._markGlassAndDependents(owner);
   166→   265→   265→                                }
   167→   266→   266→                        }
   168→   267→   267→                });

   170→   269→   269→                        this._glassSubtreeObserver.observe(el, {
   171→   270→   270→                                childList: true,
   172→   271→   271→                                subtree: true,
   173→   272→   272→                                characterData: true,
   174→   273→   273→                                attributes: true,
   175→   274→   274→                                attributeFilter: ['data-config'],
   176→   275→   275→                        });
   177→   276→   276→                }
   178→   277→   277→                this._glassContentDirty.clear();
   179→   278→   278→
   180→   279→   279→                this._running = true;
   181→   280→   280→                this._globalDirty = true;
   182→   281→   281→                this._rafId = requestAnimationFrame(() => this._renderLoop());
   183→   282→   282→        }
   184→   283→   283→

   230→   329→   329→        private _setupGlassElements(): void {
   231→   330→   330→                let needsButtonStyles = false;
   232→   331→   331→
   233→   332→   332→                for (const el of this.glassSet) {
   234→   333→   333→                        // Glass elements must be direct children of the root.
   235→   334→   334→                        if (el.parentElement !== this.root) {
   236→   335→   335→                                console.warn('LiquidGlass: glass element must be a direct child of root, skipping.', el);
   237→   336→   336→                                this.glassSet.delete(el);
   238→   337→   337→                                continue;
   239→   338→   338→                        }
   240→   339→   339→
   241→   340→   340→                        const currentPosition = window.getComputedStyle(el).position;
   242→   341→   341→                        if (currentPosition === 'static') {
   243→   342→   342→                                el.style.position = 'relative';
   244→   343→   343→                        }
   245→   344→   344→                        el.style.overflow = 'visible';
   246→   345→   345→
   247→   346→   346→                        const config = this._getConfig(el);
   248→   347→   347→

   250→   349→   349→                        // scroll/pan on floating (draggable) glass elements.
   251→   350→   350→                        if (config.floating) {
   252→   351→   351→                                el.style.touchAction = 'none';
   253→   352→   352→                        }
   254→   353→   353→
   255→   354→   354→                        // Button mode — cursor + hover/press shader-state listeners
   256→   355→   355→                        if (config.button) {
   257→   356→   356→                                el.classList.add(BUTTON_CLASS);
   258→   357→   357→                                needsButtonStyles = true;

   260→   359→   359→                        }
   261→   360→   360→
   262→   361→   361→                        const canvas = document.createElement('canvas');
   263→   362→   362→                        canvas.style.cssText = 'position:absolute;inset:0;width:100%;height:100%;pointer-events:none;z-index:-1;';
   264→   363→   363→                        el.insertBefore(canvas, el.firstChild);
   265→   364→   364→
   266→   365→   365→                        this.glassCanvases.set(el, canvas);
   267→   366→   366→                }
   268→   367→   367→
   269→   368→   368→                // Inject button styles once if any glass element uses button mode
   270→   369→   369→                if (needsButtonStyles && !document.getElementById(STYLE_ID)) {
   271→   370→   370→                        const style = document.createElement('style');
   272→   371→   371→                        style.id = STYLE_ID;
   273→   372→   372→                        style.textContent = BUTTON_CSS;
   274→   373→   373→                        document.head.appendChild(style);
   275→   374→   374→                }
   276→   375→   375→
   277→   376→   376→        }
   278→   377→   377→
   279→   378→   378→        /**
   280→   379→   379→         * Walk up from a mutation target until we hit a glass element
   281→   380→   380→         * registered on this instance. Returns null if the node isn't
   282→   381→   381→         * inside any glass subtree (shouldn't normally happen since the
   283→   382→   382→         * observers are scoped to glass elements, but the mutation target
   284→   383→   383→         * may be a Text node or detached during a removal).
   285→   384→   384→         */
   286→   385→   385→        private _closestGlassAncestor(node: Node): HTMLElement | null {
   287→   386→   386→                let cur: Node | null = node;
   288→   387→   387→                while (cur) {
   289→   388→   388→                        if (cur.nodeType === 1 && this.glassSet.has(cur as HTMLElement)) {
   290→   389→   389→                                return cur as HTMLElement;
   291→   390→   390→                        }
   292→   391→   391→                        cur = cur.parentNode;
   293→   392→   392→                }
   294→   393→   393→                return null;
   295→   394→   394→        }
   296→   395→   395→
   297→   396→   396→        /**
   298→   397→   397→         * Mark a glass element (and any glass that visually depends on it
   299→   398→   398→         * via z-order overlap) as needing a shader re-render on the next
   300→   399→   399→         * frame.
   301→   400→   400→         *
   302→   401→   401→         * `rectOverride` lets callers pass a rect that differs from the
   303→   402→   402→         * element's current bounding box — useful for drag, where we
   304→   403→   403→         * want to invalidate both the *old* and *new* footprints in the
   305→   404→   404→         * same call so glasses behind the dragged panel can clear its
   306→   405→   405→         * trail and glasses ahead can pick up its new shadow.
   307→   406→   406→         */
   308→   407→   407→        private _markGlassAndDependents(
   309→   408→   408→                element: HTMLElement,
   310→   409→   409→                rectOverride?: DOMRect,
   311→   410→   410→        ): void {
   312→   411→   411→                // 1. The element itself, if it's a glass.
   313→   412→   412→                if (this.glassSet.has(element)) {
   314→   413→   413→                        this._glassDirty.add(element);
   315→   414→   414→                }
   316→   415→   415→
   317→   416→   416→                // 2. Glass elements rendered AFTER `element` in the painting
   318→   417→   417→                //    order whose sample rect intersects element's bounds.
   319→   418→   418→                //    They read element via _composeSceneForGlass (either as a
   320→   419→   419→                //    non-glass contributor or via _drawPriorGlassToScene).
   321→   420→   420→                const rootRect = this.root.getBoundingClientRect();
   322→   421→   421→                const dpr = window.devicePixelRatio || 1;
   323→   422→   422→                const elementDOMRect = rectOverride ?? element.getBoundingClientRect();
   324→   423→   423→                // Pad by SHADOW_PAD when the element itself is a glass — its
   325→   424→   424→                // rendered output extends that far past its CSS box. For non-
   326→   425→   425→                // glass contributors the bounds are exact (their paint overflow
   327→   426→   426→                // is handled by `_getPaintOverflowPad` at draw time, which is
   328→   427→   427→                // fine for visual fidelity but not relevant to dirty marking).
   329→   428→   428→                const elementBox = this._getPixelRect(
   330→   429→   429→                        elementDOMRect,
   331→   430→   430→                        rootRect,
   332→   431→   431→                        dpr,
   333→   432→   432→                        this.glassSet.has(element) ? SHADOW_PAD : 0,
   334→   433→   433→                );
   335→   434→   434→
   336→   435→   435→                let seenElement = false;
   337→   436→   436→                for (const child of this._sortedChildren) {
   338→   437→   437→                        if (child === element) { seenElement = true; continue; }
   339→   438→   438→                        if (!seenElement) continue;
   340→   439→   439→                        if (!this.glassSet.has(child)) continue;
   341→   440→   440→                        const sampleRect = this._getPixelRect(
   342→   441→   441→                                child.getBoundingClientRect(), rootRect, dpr, SHADOW_PAD,
   343→   442→   442→                        );
   344→   443→   443→                        if (LiquidGlass._rectsIntersect(elementBox, sampleRect)) {
   345→   444→   444→                                this._glassDirty.add(child);
   346→   445→   445→                        }
   347→   446→   446→                }
   348→   447→   447→        }
   349→   448→   448→
   350→   449→   449→        /**
   351→   450→   450→         * Mark every glass element whose sample rect intersects the given
   352→   451→   451→         * element's bounding rect, regardless of stacking order. Used by
   353→   452→   452→         * the async cache-update callback (a wrapper's pixels just got
   354→   453→   453→         * fresh, so any glass that samples them needs to re-render) and
   355→   454→   454→         * by the public markChanged() API for elements outside the glass
   356→   455→   455→         * set.
   357→   456→   456→         */
   358→   457→   457→        private _markGlassesIntersecting(element: HTMLElement): void {
   359→   458→   458→                const rootRect = this.root.getBoundingClientRect();
   360→   459→   459→                const dpr = window.devicePixelRatio || 1;
   361→   460→   460→                const elementBox = this._getPixelRect(
   362→   461→   461→                        element.getBoundingClientRect(), rootRect, dpr,
   363→   462→   462→                        this.glassSet.has(element) ? SHADOW_PAD : 0,
   364→   463→   463→                );
   365→   464→   464→                for (const glass of this.glassSet) {
   366→   465→   465→                        const sampleRect = this._getPixelRect(
   367→   466→   466→                                glass.getBoundingClientRect(), rootRect, dpr, SHADOW_PAD,
   368→   467→   467→                        );
   369→   468→   468→                        if (LiquidGlass._rectsIntersect(elementBox, sampleRect)) {
   370→   469→   469→                                this._glassDirty.add(glass);
   371→   470→   470→                        }
   372→   471→   471→                }
   373→   472→   472→        }
   374→   473→   473→
   375→   474→   474→        /**
   376→   475→   475→         * Public API: mark an element (or all glass elements when called
   377→   476→   476→         * with no arguments) as needing a shader re-render on the next
   378→   477→   477→         * frame. Useful for content the library can't observe on its own —
   379→   478→   478→         * a `<canvas>` whose pixels you just updated, an `<img>` you just
   380→   479→   479→         * swapped via JS, a wrapper whose CSS background-image you just
   381→   480→   480→         * changed, etc.
   382→   481→   481→         *
   383→   482→   482→         * For elements registered via `data-dynamic`, the library already
   384→   483→   483→         * treats them as always-dirty and re-renders affected glasses
   385→   484→   484→         * every frame; calling markChanged() on them is a no-op but is
   386→   485→   485→         * harmless.
   387→   486→   486→         *
   388→   487→   487→         * @param element The element that changed visually. Pass nothing
   389→   488→   488→         * (or `undefined`) to mark every glass on this instance dirty.
   390→   489→   489→         */
   391→   490→   490→        markChanged(element?: HTMLElement): void {
   392→   491→   491→                if (!element) {
   393→   492→   492→                        this._globalDirty = true;
   394→   493→   493→                        return;

   399→   498→   498→        private _setupButtonListeners(el: HTMLElement): void {
   400→   499→   499→                const state: ButtonState = { hover: false, pressed: false };
   401→   500→   500→                this._buttonStates.set(el, state);
   402→   501→   501→
   403→   502→   502→                // Button state change just affects this button's shader uniforms

   409→   508→   508→                const onDown = () => { state.pressed = true; mark(); };

   414→   513→   513→                el.addEventListener('pointerdown', onDown);
   415→   514→   514→                el.addEventListener('pointerup', onUp);
   416→   515→   515→                el.addEventListener('pointercancel', onUp);
   417→   516→   516→

   419→   518→   518→                        () => el.removeEventListener('pointerover', onOver),
   420→   519→   519→                        () => el.removeEventListener('pointerout', onOut),
   421→   520→   520→                        () => el.removeEventListener('pointerdown', onDown),
   422→   521→   521→                        () => el.removeEventListener('pointerup', onUp),
   423→   522→   522→                        () => el.removeEventListener('pointercancel', onUp),

   425→   524→   524→        }

   429→   528→   528→        // ────────────────────────────────────────────

   431→   530→   530→        /**
   432→   531→   531→         * Re-capture the DOM content (text, icons, etc.) of glass elements
   433→   532→   532→         * whose subtrees have been mutated since the last capture, hiding
   434→   533→   533→         * the injected shader canvas so it isn't included.
   435→   534→   534→         *
   436→   535→   535→         * Pass `targets = null` to capture every glass element (used at
   437→   536→   536→         * init and on resize); pass a Set to capture only specific ones.
   438→   537→   537→         *
   439→   538→   538→         * Guarded against concurrent execution: if a capture is already
   440→   539→   539→         * running, the affected elements stay in `_glassContentDirty` and
   441→   540→   540→         * the next render-loop tick picks them up.
   442→   541→   541→         */
   443→   542→   542→        private async _captureGlassContent(
   444→   543→   543→                targets: Set<HTMLElement> | null = null,
   445→   544→   544→        ): Promise<void> {
   446→   545→   545→                if (this._capturingGlassContent) return;
   447→   546→   546→                this._capturingGlassContent = true;
   448→   547→   547→                try {
   449→   548→   548→                        for (const [el, glassCanvas] of this.glassCanvases) {
   450→   549→   549→                                if (targets && !targets.has(el)) continue;
   451→   550→   550→                                const rect = el.getBoundingClientRect();
   452→   551→   551→                                const img = await this.capture.captureToCanvas(
   453→   552→   552→                                        el,
   454→   553→   553→                                        rect.width,
   455→   554→   554→                                        rect.height,
   456→   555→   555→                                        [glassCanvas],
   457→   556→   556→                                );
   458→   557→   557→                                if (img) {
   459→   558→   558→                                        this._glassContentImages.set(el, img);
   460→   559→   559→                                }
   461→   560→   560→                        }
   462→   561→   561→                } finally {
   463→   562→   562→                        this._capturingGlassContent = false;
   464→   563→   563→                }
   465→   564→   564→        }
   466→   565→   565→
   467→   566→   566→        /**
   468→   567→   567→         * Synchronously walk every non-glass direct child of root and
   469→   568→   568→         * await its html-to-image capture so the cache is fully populated
   470→   569→   569→         * by the time the render loop starts. Without this, the first
   471→   570→   570→         * frame's glass shader sees an empty (white) local scene for
   472→   571→   571→         * ~one or two frames while the async captures resolve.
   473→   572→   572→         */
   474→   573→   573→        private async _prewarmStaticCaptures(): Promise<void> {
   475→   574→   574→                for (const child of this._sortedChildren) {
   476→   575→   575→                        if (this.glassSet.has(child)) continue;

   478→   577→   577→                        if (tag === 'CANVAS' || tag === 'IMG' || tag === 'VIDEO') continue;
   479→   578→   578→                        if (child.hasAttribute('data-dynamic')) continue;
   480→   579→   579→                        try {
   481→   580→   580→                                await this.capture.captureElement(child, false);
   482→   581→   581→                        } catch (err) {
   483→   582→   582→                                console.warn('LiquidGlass: prewarm capture failed:', child, err);
   484→   583→   583→                        }
   485→   584→   584→                }
   486→   585→   585→        }

   490→   589→   589→        // ────────────────────────────────────────────

   492→   591→   591→        private _getSortedChildren(): HTMLElement[] {
   493→   592→   592→                const children = Array.from(this.root.children) as HTMLElement[];
   494→   593→   593→                const rootDisplay = window.getComputedStyle(this.root).display;
   495→   594→   594→                const isFlexOrGridParent =
   496→   595→   595→                        rootDisplay === 'flex' || rootDisplay === 'inline-flex' ||
   497→   596→   596→                        rootDisplay === 'grid' || rootDisplay === 'inline-grid';
   498→   597→   597→
   499→   598→   598→                const tagged = children.map((el, domIndex) => {
   500→   599→   599→                        const style = window.getComputedStyle(el);
   501→   600→   600→                        const hasStackingContext =
   502→   601→   601→                                LiquidGlass._formsStackingContext(style, isFlexOrGridParent);
   503→   602→   602→                        const rawZ = parseInt(style.zIndex, 10);
   504→   603→   603→                        const zIndex = isNaN(rawZ) ? 0 : rawZ;
   505→   604→   604→                        return { el, domIndex, hasStackingContext, zIndex };
   506→   605→   605→                });
   507→   606→   606→
   508→   607→   607→                tagged.sort((a, b) => {
   509→   608→   608→                        if (!a.hasStackingContext && b.hasStackingContext) return -1;
   510→   609→   609→                        if (a.hasStackingContext && !b.hasStackingContext) return 1;
   511→   610→   610→                        if (a.hasStackingContext && b.hasStackingContext) {
   512→   611→   611→                                if (a.zIndex !== b.zIndex) return a.zIndex - b.zIndex;
   513→   612→   612→                        }
   514→   613→   613→                        return a.domIndex - b.domIndex;
   515→   614→   614→                });
   516→   615→   615→
   517→   616→   616→                return tagged.map(t => t.el);
   518→   617→   617→        }
   519→   618→   618→
   520→   619→   619→        /**
   521→   620→   620→         * Returns true when the element forms a CSS stacking context — i.e.
   522→   621→   621→         * when its z-index participates in painting order. Mirrors the spec:
   523→   622→   622→         * https://developer.mozilla.org/en-US/docs/Web/CSS/CSS_positioned_layout/Stacking_context
   524→   623→   623→         *
   525→   624→   624→         * Used by `_getSortedChildren` to decide painting order for the
   526→   625→   625→         * local scene assembly. The set of triggers needs to match the
   527→   626→   626→         * browser's actual stacking model — otherwise overlays end up
   528→   627→   627→         * painted before the background image and get erased.
   529→   628→   628→         */
   530→   629→   629→        private static _formsStackingContext(
   531→   630→   630→                style: CSSStyleDeclaration,
   532→   631→   631→                isFlexOrGridParent: boolean,
   533→   632→   632→        ): boolean {
   534→   633→   633→                if (style.position !== 'static') return true;
   535→   634→   634→                if (isFlexOrGridParent && style.zIndex !== 'auto') return true;
   536→   635→   635→                if (parseFloat(style.opacity) < 1) return true;
   537→   636→   636→                if (style.transform !== 'none' && style.transform !== '') return true;
   538→   637→   637→                if (style.filter !== 'none' && style.filter !== '') return true;
   539→   638→   638→                if (style.perspective !== 'none' && style.perspective !== '') return true;
   540→   639→   639→                if (style.clipPath !== 'none' && style.clipPath !== '') return true;
   541→   640→   640→                if (style.mixBlendMode !== 'normal' && style.mixBlendMode !== '') return true;
   542→   641→   641→                if (style.isolation === 'isolate') return true;
   543→   642→   642→
   544→   643→   643→                const bf = style.backdropFilter
   545→   644→   644→                        || (style as unknown as { webkitBackdropFilter?: string }).webkitBackdropFilter;
   546→   645→   645→                if (bf && bf !== 'none') return true;
   547→   646→   646→
   548→   647→   647→                const mask = style.maskImage
   549→   648→   648→                        || (style as unknown as { webkitMaskImage?: string }).webkitMaskImage;
   550→   649→   649→                if (mask && mask !== 'none') return true;
   551→   650→   650→
   552→   651→   651→                const contain = style.contain;
   553→   652→   652→                if (contain && /\b(layout|paint|strict|content)\b/.test(contain)) return true;
   554→   653→   653→
   555→   654→   654→                if (style.willChange) {
   556→   655→   655→                        const triggers = new Set([
   557→   656→   656→                                'transform', 'opacity', 'filter', 'backdrop-filter',
   558→   657→   657→                                'perspective', 'clip-path', 'mask', 'mask-image',
   559→   658→   658→                                'isolation', 'mix-blend-mode',
   560→   659→   659→                        ]);
   561→   660→   660→                        const tokens = style.willChange.split(',').map(t => t.trim());
   562→   661→   661→                        for (const t of tokens) {
   563→   662→   662→                                if (triggers.has(t)) return true;
   564→   663→   663→                        }
   565→   664→   664→                }
   566→   665→   665→
   567→   666→   666→                return false;
   568→   667→   667→        }

   594→   693→   693→                const cachedEl = el as ConfigCachedElement;
   595→   694→   694→                const configKey = el.dataset.config ?? '';
   596→   695→   695→
   597→   696→   696→                if (cachedEl.configCacheKey !== configKey) {

   602→   701→   701→                                        if (parsed && typeof parsed === 'object') {
   603→   702→   702→                                                perElement = parsed as Partial<GlassConfig>;
   604→   703→   703→                                        } else {
   605→   704→   704→                                                console.warn('LiquidGlass: data-config must decode to an object for element:', el);
   606→   705→   705→                                        }
   607→   706→   706→                                } catch (_e) {

   609→   708→   708→                                }

   612→   711→   711→                        cachedEl.configCacheKey = configKey;
   613→   712→   712→                }
   614→   713→   713→

   616→   715→   715→

   619→   718→   718→                        if (state) {
   620→   719→   719→                                if (state.pressed) {
   621→   720→   720→                                        config.zRadius = config.zRadius * 0.8;
   622→   721→   721→                                        config.shadowSpread = config.shadowSpread * 1.2;
   623→   722→   722→                                        // brightness reset to original (no hover boost)
   624→   723→   723→                                } else if (state.hover) {
   625→   724→   724→                                        config.brightness = config.brightness + 0.2;
   626→   725→   725→                                }
   627→   726→   726→                        }
   628→   727→   727→                }
   629→   728→   728→
   630→   729→   729→                return config;
   631→   730→   730→        }

   655→   754→   754→        private _updateGlassCanvasSize(el: HTMLElement): void {
   656→   755→   755→                const canvas = this.glassCanvases.get(el);
   657→   756→   756→                if (!canvas) return;
   658→   757→   757→                const dpr = window.devicePixelRatio || 1;
   659→   758→   758→                // Use offsetWidth/Height — the CSS box size before transforms.
   660→   759→   759→                // This prevents button hover scale from inflating the canvas.
   661→   760→   760→                const elW = Math.round(el.offsetWidth);
   662→   761→   761→                const elH = Math.round(el.offsetHeight);
   663→   762→   762→                const padW = SHADOW_PAD * 2;
   664→   763→   763→                const padH = SHADOW_PAD * 2;
   665→   764→   764→                canvas.width = Math.round((elW + padW) * dpr);
   666→   765→   765→                canvas.height = Math.round((elH + padH) * dpr);
   667→   766→   766→                canvas.style.cssText = [
   668→   767→   767→                        'position:absolute',
   669→   768→   768→                        `left:${-SHADOW_PAD}px`,
   670→   769→   769→                        `top:${-SHADOW_PAD}px`,
   671→   770→   770→                        `width:${elW + padW}px`,
   672→   771→   771→                        `height:${elH + padH}px`,
   673→   772→   772→                        'pointer-events:none',
   674→   773→   773→                        'z-index:-1'
   675→   774→   774→                ].join(';') + ';';
   676→   775→   775→                this._glassLastSize.set(el, { w: elW, h: elH });
   677→   776→   776→        }

   679→   778→   778→        private _checkGlassSizeChanges(): boolean {
   680→   779→   779→                let changed = false;
   681→   780→   780→                for (const el of this.glassSet) {
   682→   781→   781→                        // Use offsetWidth/Height instead of getBoundingClientRect so
   683→   782→   782→                        // CSS transforms (e.g. button hover scale) don't trigger
   684→   783→   783→                        // false size-change detections and render loops.
   685→   784→   784→                        const w = el.offsetWidth;
   686→   785→   785→                        const h = el.offsetHeight;
   687→   786→   786→                        const last = this._glassLastSize.get(el);
   688→   787→   787→                        if (!last
   689→   788→   788→                                || Math.abs(last.w - w) > 0.5
   690→   789→   789→                                || Math.abs(last.h - h) > 0.5
   691→   790→   790→                        ) {
   692→   791→   791→                                this._updateGlassCanvasSize(el);
   693→   792→   792→                                this._glassCache.delete(el);
   694→   793→   793→                                this.capture.invalidateCache(el);
   695→   794→   794→                                // Only this element's content image needs to be
   696→   795→   795→                                // re-captured, not every glass on the page.
   697→   796→   796→                                this._glassContentDirty.add(el);
   698→   797→   797→                                changed = true;
   699→   798→   798→                        }
   700→   799→   799→                }
   701→   800→   800→                return changed;
   702→   801→   801→        }

   708→   807→   807→        /** Parse the current translate(x, y) values from an element's transform. */
   709→   808→   808→        private static _getTranslateXY(el: HTMLElement): [number, number] {
   710→   809→   809→                const style = getComputedStyle(el);
   711→   810→   810→                const matrix = style.transform;
   712→   811→   811→                if (!matrix || matrix === 'none') return [0, 0];
   713→   812→   812→                // matrix(a, b, c, d, tx, ty)
   714→   813→   813→                const m = matrix.match(/matrix\(([^)]+)\)/);
   715→   814→   814→                if (m) {
   716→   815→   815→                        const parts = m[1].split(',').map(Number);
   717→   816→   816→                        return [parts[4] || 0, parts[5] || 0];
   718→   817→   817→                }
   719→   818→   818→                return [0, 0];
   720→   819→   819→        }

   722→   821→   821→        private _handlePointerDown(e: PointerEvent): void {
   723→   822→   822→                // Iterate all glass elements in reverse stacking order (topmost first).
   724→   823→   823→                for (let i = this._sortedChildren.length - 1; i >= 0; i--) {
   725→   824→   824→                        const el = this._sortedChildren[i];
   726→   825→   825→                        if (!this.glassSet.has(el)) continue;
   727→   826→   826→
   728→   827→   827→                        const config = this._getConfig(el);
   729→   828→   828→                        if (!config.floating) continue;
   730→   829→   829→
   731→   830→   830→                        const rect = el.getBoundingClientRect();
   732→   831→   831→                        // Use the CSS box size (offsetWidth/Height) for hit testing,
   733→   832→   832→                        // but use the bounding rect position (which is correct for
   734→   833→   833→                        // elements positioned via CSS, grid, etc.).
   735→   834→   834→                        const elW = el.offsetWidth;
   736→   835→   835→                        const elH = el.offsetHeight;
   737→   836→   836→                        // The visual position includes the shadow canvas overflow.
   738→   837→   837→                        // Compute the element's true visual origin by centering the
   739→   838→   838→                        // offset size within the bounding rect.
   740→   839→   839→                        const visualLeft = rect.left + (rect.width - elW) / 2;
   741→   840→   840→                        const visualTop = rect.top + (rect.height - elH) / 2;
   742→   841→   841→
   743→   842→   842→                        if (
   744→   843→   843→                                e.clientX >= visualLeft && e.clientX <= visualLeft + elW &&
   745→   844→   844→                                e.clientY >= visualTop && e.clientY <= visualTop + elH
   746→   845→   845→                        ) {
   747→   846→   846→                                const [tx, ty] = LiquidGlass._getTranslateXY(el);
   748→   847→   847→                                this._drag.active = true;
   749→   848→   848→                                this._drag.element = el;
   750→   849→   849→                                this._drag.startX = e.clientX;
   751→   850→   850→                                this._drag.startY = e.clientY;
   752→   851→   851→                                this._drag.origTx = tx;
   753→   852→   852→                                this._drag.origTy = ty;
   754→   853→   853→                                el.style.cursor = 'grabbing';
   755→   854→   854→                                el.setPointerCapture(e.pointerId);
   756→   855→   855→                                e.preventDefault();
   757→   856→   856→                                break;
   758→   857→   857→                        }
   759→   858→   858→                }
   760→   859→   859→        }
   761→   860→   860→
   762→   861→   861→        private _handlePointerMove(e: PointerEvent): void {
   763→   862→   862→                if (!this._drag.active) {
   764→   863→   863→                        for (const el of this.glassSet) {
   765→   864→   864→                                const config = this._getConfig(el);
   766→   865→   865→                                if (!config.floating) continue;
   767→   866→   866→                                const rect = el.getBoundingClientRect();
   768→   867→   867→                                const elW = el.offsetWidth;
   769→   868→   868→                                const elH = el.offsetHeight;
   770→   869→   869→                                const visualLeft = rect.left + (rect.width - elW) / 2;
   771→   870→   870→                                const visualTop = rect.top + (rect.height - elH) / 2;
   772→   871→   871→                                if (
   773→   872→   872→                                        e.clientX >= visualLeft && e.clientX <= visualLeft + elW &&
   774→   873→   873→                                        e.clientY >= visualTop && e.clientY <= visualTop + elH
   775→   874→   874→                                ) {
   776→   875→   875→                                        el.style.cursor = 'grab';
   777→   876→   876→                                } else {
   778→   877→   877→                                        el.style.cursor = '';
   779→   878→   878→                                }
   780→   879→   879→                        }
   781→   880→   880→                        return;
   782→   881→   881→                }
   783→   882→   882→
   784→   883→   883→                const el = this._drag.element!;
   785→   884→   884→                const dx = e.clientX - this._drag.startX;
   786→   885→   885→                const dy = e.clientY - this._drag.startY;
   787→   886→   886→                let newTx = this._drag.origTx + dx;
   788→   887→   887→                let newTy = this._drag.origTy + dy;
   789→   888→   888→
   790→   889→   889→                // Constrain within root bounds with margin.
   791→   890→   890→                // For nested elements, offsetLeft/Top is relative to offsetParent
   792→   891→   891→                // (which may not be root). Use getBoundingClientRect to compute
   793→   892→   892→                // the element's position relative to root, then subtract the
   794→   893→   893→                // current translate to get the base (CSS layout) position.
   795→   894→   894→                const rootRect = this.root.getBoundingClientRect();
   796→   895→   895→                const elW = el.offsetWidth;
   797→   896→   896→                const elH = el.offsetHeight;
   798→   897→   897→                const elRect = el.getBoundingClientRect();
   799→   898→   898→                const [curTx, curTy] = LiquidGlass._getTranslateXY(el);
   800→   899→   899→                const baseLeft = (elRect.left + (elRect.width - elW) / 2) - rootRect.left - curTx;
   801→   900→   900→                const baseTop = (elRect.top + (elRect.height - elH) / 2) - rootRect.top - curTy;
   802→   901→   901→                const margin = 10;
   803→   902→   902→                const posLeft = baseLeft + newTx;
   804→   903→   903→                const posTop = baseTop + newTy;
   805→   904→   904→                const maxLeft = rootRect.width - elW - margin;
   806→   905→   905→                const maxTop = rootRect.height - elH - margin;
   807→   906→   906→                if (posLeft < margin) newTx += margin - posLeft;
   808→   907→   907→                if (posTop < margin) newTy += margin - posTop;
   809→   908→   908→                if (posLeft > maxLeft) newTx -= posLeft - maxLeft;
   810→   909→   909→                if (posTop > maxTop) newTy -= posTop - maxTop;
   811→   910→   910→
   812→   911→   911→                // Mark glasses overlapping the OLD position so they re-render
   813→   912→   912→                // to clear the dragged element's previous footprint, THEN move
   814→   913→   913→                // the element, THEN mark glasses overlapping the NEW position
   815→   914→   914→                // so they pick up its new footprint.
   816→   915→   915→                const oldRect = el.getBoundingClientRect();
   817→   916→   916→                this._markGlassAndDependents(el, oldRect);
   818→   917→   917→                el.style.transform = `translate(${newTx}px, ${newTy}px)`;
   819→   918→   918→                this._markGlassAndDependents(el);
   820→   919→   919→        }
   821→   920→   920→
   822→   921→   921→        private _handlePointerUp(_e: PointerEvent): void {
   823→   922→   922→                if (!this._drag.active) return;
   824→   923→   923→                const dragged = this._drag.element!;
   825→   924→   924→                dragged.style.cursor = '';
   826→   925→   925→                this._drag.active = false;
   827→   926→   926→                this._drag.element = null;
   828→   927→   927→                // `isDragging` flips off, so the per-glass gate stops forcing
   829→   928→   928→                // every-frame re-renders for this glass and its overlapping
   830→   929→   929→                // neighbours. Mark them once so they get a final clean render.
   831→   930→   930→                this._markGlassAndDependents(dragged);
   832→   931→   931→        }

   834→   933→   933→        // ────────────────────────────────────────────
   835→   934→   934→        // Render loop
   836→   935→   935→        // ────────────────────────────────────────────
   837→   936→   936→
   838→   937→   937→        private _renderLoop(): void {
   839→   938→   938→                if (!this._running) return;
   840→   939→   939→
   841→   940→   940→                // FPS tracking
   842→   941→   941→                const now = performance.now();
   843→   942→   942→                this._fpsFrames++;
   844→   943→   943→                if (now - this._fpsTime >= 1000) {
   845→   944→   944→                        this.fps = this._fpsFrames;
   846→   945→   945→                        this._fpsFrames = 0;
   847→   946→   946→                        this._fpsTime = now;
   848→   947→   947→                }
   849→   948→   948→
   850→   949→   949→                if (this._checkGlassSizeChanges()) {
   851→   950→   950→                        // _checkGlassSizeChanges already added the resized
   852→   951→   951→                        // elements to _glassDirty per-element; nothing more to do.
   853→   952→   952→                }
   854→   953→   953→
   855→   954→   954→                if (this._glassContentDirty.size > 0 && !this._capturingGlassContent) {
   856→   955→   955→                        // Snapshot the dirty set before draining: any mutations
   857→   956→   956→                        // that arrive while the async capture is in flight stay
   858→   957→   957→                        // in the live set and are picked up on the next tick.
   859→   958→   958→                        const targets = new Set(this._glassContentDirty);
   860→   959→   959→                        this._glassContentDirty.clear();
   861→   960→   960→                        this._captureGlassContent(targets);
   862→   961→   961→                }
   863→   962→   962→
   864→   963→   963→                try {
   865→   964→   964→                        this._renderFrame();
   866→   965→   965→                } catch (err) {
   867→   966→   966→                        console.error('LiquidGlass: render error:', err);
   868→   967→   967→                }
   869→   968→   968→
   870→   969→   969→                this._rafId = requestAnimationFrame(() => this._renderLoop());
   871→   970→   970→        }
   872→   971→   971→
   873→   972→   972→        private _renderFrame(): void {
   874→   973→   973→                const dpr = window.devicePixelRatio || 1;
   875→   974→   974→                const rootRect = this.root.getBoundingClientRect();
   876→   975→   975→                const isDragging = this._drag.active;
   877→   976→   976→
   878→   977→   977→                // 1. Fan out user markChanged() calls into per-glass dirty marks.
   879→   978→   978→                if (this._userMarkedChanged.size > 0) {
   880→   979→   979→                        for (const el of this._userMarkedChanged) {
   881→   980→   980→                                this._markGlassesIntersecting(el);
   882→   981→   981→                        }
   883→   982→   982→                        this._userMarkedChanged.clear();
   884→   983→   983→                }
   885→   984→   984→
   886→   985→   985→                // 2. Promote any global dirty into per-element dirties so the
   887→   986→   986→                //    rest of the loop only ever consults `_glassDirty`.
   888→   987→   987→                if (this._globalDirty) {
   889→   988→   988→                        for (const el of this.glassSet) this._glassDirty.add(el);
   890→   989→   989→                        this._globalDirty = false;
   891→   990→   990→                }
   892→   991→   991→
   893→   992→   992→                const needsRender = this._glassDirty.size > 0
   894→   993→   993→                        || this._hasDynamic
   895→   994→   994→                        || isDragging;
   896→   995→   995→                if (!needsRender) return;
   897→   996→   996→
   898→   997→   997→                // 3. Snapshot + drain the dirty set so anything added during
   899→   998→   998→                //    this frame's work (e.g. async cache landings) is picked
   900→   999→   999→                //    up on the next frame instead of getting clobbered.
   901→  1000→  1000→                const dirtyTargets = new Set(this._glassDirty);
   902→  1001→  1001→                this._glassDirty.clear();
   903→  1002→  1002→
   904→  1003→  1003→                // 4. Track which glass elements actually re-rendered this
   905→  1004→  1004→                //    frame, with their sample rect, so later glasses in the
   906→  1005→  1005→                //    z-order can detect "a prior glass that I overlap just
   907→  1006→  1006→                //    re-rendered → I need to refresh too."
   908→  1007→  1007→                const renderedThisFrame: Array<{ rect: SampleRect }> = [];
   909→  1008→  1008→
   910→  1009→  1009→                for (const child of this._sortedChildren) {
   911→  1010→  1010→                        if (!this.glassSet.has(child)) continue;
   912→  1011→  1011→                        this._renderGlassElement(
   913→  1012→  1012→                                child,
   914→  1013→  1013→                                rootRect,
   915→  1014→  1014→                                dpr,
   916→  1015→  1015→                                isDragging,
   917→  1016→  1016→                                dirtyTargets,
   918→  1017→  1017→                                renderedThisFrame,
   919→  1018→  1018→                        );
   920→  1019→  1019→                }
   921→  1020→  1020→        }
   922→  1021→  1021→
   923→  1022→  1022→        /**
   924→  1023→  1023→         * Render a single glass element by composing just the scene region
   925→  1024→  1024→         * that can affect it, then running the shader over that local input.
   926→  1025→  1025→         *
   927→  1026→  1026→         * Whether the shader actually re-runs depends on:
   928→  1027→  1027→         *   - explicit dirty mark for this element (in `dirtyTargets`),
   929→  1028→  1028→         *   - any earlier glass in z-order that re-rendered this frame
   930→  1029→  1029→         *     and whose rect intersects this glass's sample rect,
   931→  1030→  1030→         *   - this glass having moved since last frame (position cache),
   932→  1031→  1031→         *   - this glass having dynamic contributors in its sample (video,
   933→  1032→  1032→         *     data-dynamic),
   934→  1033→  1033→         *   - or active drag involving this element.
   935→  1034→  1034→         *
   936→  1035→  1035→         * On render, an entry is pushed to `renderedThisFrame` so later
   937→  1036→  1036→         * glasses can check whether they need to refresh too.
   938→  1037→  1037→         */
   939→  1038→  1038→        private _renderGlassElement(
   940→  1039→  1039→                child: HTMLElement,
   941→  1040→  1040→                rootRect: DOMRect,
   942→  1041→  1041→                dpr: number,
   943→  1042→  1042→                isDragging: boolean,
   944→  1043→  1043→                dirtyTargets: Set<HTMLElement>,
   945→  1044→  1044→                renderedThisFrame: Array<{ rect: SampleRect }>,
   946→  1045→  1045→        ): void {
   947→  1046→  1046→                const config = this._getConfig(child);
   948→  1047→  1047→                const elRect = child.getBoundingClientRect();
   949→  1048→  1048→                const elW = child.offsetWidth;
   950→  1049→  1049→                const elH = child.offsetHeight;
   951→  1050→  1050→                const centerX = (elRect.left - rootRect.left) + elRect.width / 2;
   952→  1051→  1051→                const centerY = (elRect.top - rootRect.top) + elRect.height / 2;
   953→  1052→  1052→                const glassCanvas = this.glassCanvases.get(child);
   954→  1053→  1053→                const isBeingDragged = isDragging && this._drag.element === child;
   955→  1054→  1054→                const sampleRect = this._getPixelRect(elRect, rootRect, dpr, SHADOW_PAD);
   956→  1055→  1055→
   957→  1056→  1056→                const cached = this._glassCache.get(child);
   958→  1057→  1057→                const posChanged = !cached
   959→  1058→  1058→                        || Math.abs(cached.centerX - centerX) > 0.5
   960→  1059→  1059→                        || Math.abs(cached.centerY - centerY) > 0.5;
   961→  1060→  1060→                const hasDynamicContributors = this._hasDynamic
   962→  1061→  1061→                        && this._glassHasDynamicContributors(child, sampleRect, rootRect, dpr);
   963→  1062→  1062→
   964→  1063→  1063→                // Did any earlier-rendered glass actually overlap this glass's
   965→  1064→  1064→                // sample rect? Replaces the old monotonic `bgChanged` boolean
   966→  1065→  1065→                // with a per-element intersection check.
   967→  1066→  1066→                let priorGlassChanged = false;
   968→  1067→  1067→                for (const r of renderedThisFrame) {
   969→  1068→  1068→                        if (LiquidGlass._rectsIntersect(r.rect, sampleRect)) {
   970→  1069→  1069→                                priorGlassChanged = true;
   971→  1070→  1070→                                break;
   972→  1071→  1071→                        }
   973→  1072→  1072→                }
   974→  1073→  1073→
   975→  1074→  1074→                const isExplicitlyDirty = dirtyTargets.has(child);
   976→  1075→  1075→
   977→  1076→  1076→                const needsShaderRender = isDragging
   978→  1077→  1077→                        ? (isBeingDragged || isExplicitlyDirty || priorGlassChanged || hasDynamicContributors)
   979→  1078→  1078→                        : (!cached || posChanged || isExplicitlyDirty || priorGlassChanged || hasDynamicContributors);
   980→  1079→  1079→
   981→  1080→  1080→                if (needsShaderRender && glassCanvas) {
   982→  1081→  1081→                        // Use glassCanvas dimensions as the authoritative size for
   983→  1082→  1082→                        // the renderer pass. sampleRect.w/h (from getBoundingClientRect)
   984→  1083→  1083→                        // can differ from glassCanvas.width/height (from offsetWidth)
   985→  1084→  1084→                        // by ±1 device pixel due to sub-pixel rounding differences.
   986→  1085→  1085→                        // If the renderer clears fewer pixels than drawImage copies,
   987→  1086→  1086→                        // the rightmost/bottom columns contain stale pixels → a thin
   988→  1087→  1087→                        // visible line. Using the same source of truth for both avoids
   989→  1088→  1088→                        // the mismatch.
   990→  1089→  1089→                        const renderW = glassCanvas.width;
   991→  1090→  1090→                        const renderH = glassCanvas.height;
   992→  1091→  1091→                        this._composeSceneForGlass(child, sampleRect, rootRect, dpr);
   993→  1092→  1092→                        this.renderer.uploadAndBlur(
   994→  1093→  1093→                                this._sceneCanvas,
   995→  1094→  1094→                                0,
   996→  1095→  1095→                                0,
   997→  1096→  1096→                                renderW,
   998→  1097→  1097→                                renderH,
   999→  1098→  1098→                                config.blurAmount,
  1000→  1099→  1099→                        );
  1001→  1100→  1100→                        this.renderer.clear();
  1002→  1101→  1101→                        this.renderer.renderGlassPanel(
  1003→  1102→  1102→                                config,
  1004→  1103→  1103→                                elW,
  1005→  1104→  1104→                                elH,
  1006→  1105→  1105→                                dpr,
  1007→  1106→  1106→                        );
  1008→  1107→  1107→
  1009→  1108→  1108→                        const ctx = glassCanvas.getContext('2d')!;
  1010→  1109→  1109→                        ctx.clearRect(0, 0, glassCanvas.width, glassCanvas.height);
  1011→  1110→  1110→                        ctx.drawImage(
  1012→  1111→  1111→                                this.renderer.canvas,
  1013→  1112→  1112→                                0, 0, glassCanvas.width, glassCanvas.height,
  1014→  1113→  1113→                                0, 0, glassCanvas.width, glassCanvas.height,
  1015→  1114→  1114→                        );
  1016→  1115→  1115→
  1017→  1116→  1116→                        this._glassCache.set(child, { centerX, centerY });
  1018→  1117→  1117→                        renderedThisFrame.push({ rect: sampleRect });
  1019→  1118→  1118→                }
  1020→  1119→  1119→        }
  1021→  1120→  1120→
  1022→  1121→  1121→        /**
  1023→  1122→  1122→         * Build the local input scene for a glass panel by walking only the
  1024→  1123→  1123→         * contributors that paint before it in the stacking order.
  1025→  1124→  1124→         */
  1026→  1125→  1125→        private _composeSceneForGlass(
  1027→  1126→  1126→                currentGlass: HTMLElement,
  1028→  1127→  1127→                sampleRect: SampleRect,
  1029→  1128→  1128→                rootRect: DOMRect,
  1030→  1129→  1129→                dpr: number,
  1031→  1130→  1130→        ): void {
  1032→  1131→  1131→                this._prepareSceneCanvas(sampleRect.w, sampleRect.h);
  1033→  1132→  1132→
  1034→  1133→  1133→                for (const child of this._sortedChildren) {
  1035→  1134→  1134→                        if (child === currentGlass) break;
  1036→  1135→  1135→                        if (this.glassSet.has(child)) {
  1037→  1136→  1136→                                this._drawPriorGlassToScene(child, sampleRect, rootRect, dpr);
  1038→  1137→  1137→                        } else {
  1039→  1138→  1138→                                this._drawNonGlassChildToScene(child, sampleRect, rootRect, dpr);
  1040→  1139→  1139→                        }
  1041→  1140→  1140→                }
  1042→  1141→  1141→        }
  1043→  1142→  1142→
  1044→  1143→  1143→        private _prepareSceneCanvas(width: number, height: number): void {
  1045→  1144→  1144→                if (this._sceneCanvas.width !== width || this._sceneCanvas.height !== height) {
  1046→  1145→  1145→                        this._sceneCanvas.width = width;
  1047→  1146→  1146→                        this._sceneCanvas.height = height;
  1048→  1147→  1147→                } else {
  1049→  1148→  1148→                        this._sceneCtx.clearRect(0, 0, width, height);
  1050→  1149→  1149→                }
  1051→  1150→  1150→                this._sceneCtx.fillStyle = '#ffffff';
  1052→  1151→  1151→                this._sceneCtx.fillRect(0, 0, width, height);
  1053→  1152→  1152→        }
  1054→  1153→  1153→
  1055→  1154→  1154→        private _glassHasDynamicContributors(
  1056→  1155→  1155→                currentGlass: HTMLElement,
  1057→  1156→  1156→                sampleRect: SampleRect,
  1058→  1157→  1157→                rootRect: DOMRect,
  1059→  1158→  1158→                dpr: number,
  1060→  1159→  1159→        ): boolean {
  1061→  1160→  1160→                // A glass element marked data-dynamic on its own root counts
  1062→  1161→  1161→                // as always-dirty: forces every-frame shader re-runs.
  1063→  1162→  1162→                if (this._childHasDynamicContent(currentGlass)) return true;
  1064→  1163→  1163→
  1065→  1164→  1164→                for (const child of this._sortedChildren) {
  1066→  1165→  1165→                        if (child === currentGlass) break;
  1067→  1166→  1166→                        if (this.glassSet.has(child)) continue;
  1068→  1167→  1167→                        if (!this._childHasDynamicContent(child)) continue;
  1069→  1168→  1168→                        if (this._childTouchesSample(child, sampleRect, rootRect, dpr)) {
  1070→  1169→  1169→                                return true;
  1071→  1170→  1170→                        }
  1072→  1171→  1171→                }
  1073→  1172→  1172→                return false;
  1074→  1173→  1173→        }
  1075→  1174→  1174→
  1076→  1175→  1175→        private _childHasDynamicContent(child: HTMLElement): boolean {
  1077→  1176→  1176→                if (child.hasAttribute('data-dynamic')) return true;
  1078→  1177→  1177→                if (child.tagName === 'VIDEO') return true;
  1079→  1178→  1178→                return child.querySelector('[data-dynamic], video') !== null;
  1080→  1179→  1179→        }
  1081→  1180→  1180→
  1082→  1181→  1181→        private _drawNonGlassChildToScene(
  1083→  1182→  1182→                child: HTMLElement,
  1084→  1183→  1183→                sampleRect: SampleRect,
  1085→  1184→  1184→                rootRect: DOMRect,
  1086→  1185→  1185→                dpr: number,
  1087→  1186→  1186→        ): void {
  1088→  1187→  1187→                const tag = child.tagName;
  1089→  1188→  1188→
  1090→  1189→  1189→                if (tag === 'CANVAS' || tag === 'IMG' || tag === 'VIDEO') {
  1091→  1190→  1190→                        this._drawMediaElement(child, this._sceneCtx, sampleRect, rootRect, dpr);
  1092→  1191→  1191→                        return;
  1093→  1192→  1192→                }
  1094→  1193→  1193→
  1095→  1194→  1194→                // Wrapper-level early-out: if the wrapper's own (paint-padded)
  1096→  1195→  1195→                // bounds don't intersect the sample rect, neither it nor its
  1097→  1196→  1196→                // descendants can affect this glass panel — skip the entire
  1098→  1197→  1197→                // querySelectorAll + html-to-image work.
  1099→  1198→  1198→                //
  1100→  1199→  1199→                // Caveat: a descendant absolutely-positioned outside the
  1101→  1200→  1200→                // wrapper's box won't be drawn here. Such descendants are
  1102→  1201→  1201→                // vanishingly rare in practice; if you hit it, give the escaped
  1103→  1202→  1202→                // element its own wrapper or a `data-dynamic` annotation.
  1104→  1203→  1203→                if (!this._elementTouchesSample(child, sampleRect, rootRect, dpr)) {
  1105→  1204→  1204→                        return;
  1106→  1205→  1205→                }
  1107→  1206→  1206→
  1108→  1207→  1207→                // Draw any live media descendants (img/video/canvas) directly
  1109→  1208→  1208→                // from their source elements via the fast `drawImage` path,
  1110→  1209→  1209→                // since html-to-image can't rasterise videos.
  1111→  1210→  1210→                this._captureMediaDescendants(child, this._sceneCtx, sampleRect, rootRect, dpr);
  1112→  1211→  1211→
  1113→  1212→  1212→                // Then composite the wrapper's HTML content via the cached
  1114→  1213→  1213→                // html-to-image snapshot. captureElement is async; concurrent
  1115→  1214→  1214→                // calls for the same element from multiple glass panels in the
  1116→  1215→  1215→                // same frame are deduped via HtmlCapture's `_capturing` set,
  1117→  1216→  1216→                // so the html-to-image pipeline runs at most once per frame
  1118→  1217→  1217→                // per element regardless of how many glasses overlap it.
  1119→  1218→  1218→                const isDynamic = child.hasAttribute('data-dynamic');
  1120→  1219→  1219→                this.capture.captureElement(child, isDynamic);

  1365→  1464→  1464→        /** Compute the source rectangle for drawImage that replicates CSS object-fit / object-position. */
  1366→  1465→  1465→        static _objectFitRect(
  1367→  1466→  1466→                natW: number,
  1368→  1467→  1467→                natH: number,
  1369→  1468→  1468→                boxW: number,
  1370→  1469→  1469→                boxH: number,
  1371→  1470→  1470→                fit: string,
  1372→  1471→  1471→                pos: string,
  1373→  1472→  1472→        ): ObjectFitRect {
  1374→  1473→  1473→                let sx = 0, sy = 0, sw = natW, sh = natH;
  1375→  1474→  1474→
  1376→  1475→  1475→                if (fit === 'fill' || (fit === 'scale-down' && natW <= boxW && natH <= boxH)) {
  1377→  1476→  1476→                        return { sx, sy, sw, sh };
  1378→  1477→  1477→                }
  1379→  1478→  1478→
  1380→  1479→  1479→                const parts = pos.split(/\s+/);
  1381→  1480→  1480→                const parseFrac = (v: string, total: number): number => {
  1382→  1481→  1481→                        if (v.endsWith('%')) return parseFloat(v) / 100;
  1383→  1482→  1482→                        return parseFloat(v) / total;
  1384→  1483→  1483→                };
  1385→  1484→  1484→                const fx = parseFrac(parts[0] || '50%', boxW);
  1386→  1485→  1485→                const fy = parseFrac(parts[1] || '50%', boxH);
  1387→  1486→  1486→
  1388→  1487→  1487→                if (fit === 'cover') {
  1389→  1488→  1488→                        const scale = Math.max(boxW / natW, boxH / natH);
  1390→  1489→  1489→                        sw = boxW / scale;
  1391→  1490→  1490→                        sh = boxH / scale;
  1392→  1491→  1491→                        sx = (natW - sw) * fx;
  1393→  1492→  1492→                        sy = (natH - sh) * fy;
  1394→  1493→  1493→                } else if (fit === 'contain' || fit === 'scale-down') {
  1395→  1494→  1494→                        return { sx: 0, sy: 0, sw: natW, sh: natH };
  1396→  1495→  1495→                } else if (fit === 'none') {
  1397→  1496→  1496→                        sw = boxW;
  1398→  1497→  1497→                        sh = boxH;
  1399→  1498→  1498→                        sx = (natW - sw) * fx;
  1400→  1499→  1499→                        sy = (natH - sh) * fy;
  1401→  1500→  1500→                }
  1402→  1501→  1501→
  1403→  1502→  1502→                sx = Math.max(0, Math.min(sx, natW - 1));
  1404→  1503→  1503→                sy = Math.max(0, Math.min(sy, natH - 1));
  1405→  1504→  1504→                sw = Math.min(sw, natW - sx);
  1406→  1505→  1505→                sh = Math.min(sh, natH - sy);
  1407→  1506→  1506→
  1408→  1507→  1507→                return { sx, sy, sw, sh };
  1409→  1508→  1508→        }
