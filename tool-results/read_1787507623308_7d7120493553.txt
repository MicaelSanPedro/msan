    48→function getStored(key: string): string | null {
    49→  if (typeof window === "undefined") return null;
    50→  try { return localStorage.getItem(key); } catch { return null; }
    51→}
    52→function setStored(key: string, val: string) {
    53→  try { localStorage.setItem(key, val); } catch { /* ignore */ }
    54→}
    55→function removeStored(key: string) {
    56→  try { localStorage.removeItem(key); } catch { /* ignore */ }
    57→}
    58→
    59→function getStorageSize(): string {
    60→  if (typeof window === "undefined") return "0 B";
    61→  try {
    62→    let total = 0;
    63→    for (const key of Object.keys(localStorage)) {
    64→      total += (localStorage.getItem(key) || "").length;
    65→    }
    66→    if (total < 1024) return `${total} B`;
    67→    return `${(total / 1024).toFixed(1)} KB`;
    68→  } catch { return "0 B"; }
    69→}
    70→function getVisitCount(): number {
    71→  try {
    72→    const stored = localStorage.getItem("techmate_visits");
    73→    if (stored) {
    74→      const count = parseInt(stored, 10) + 1;
    75→      localStorage.setItem("techmate_visits", String(count));
    76→      return count;
    77→    }
    78→    localStorage.setItem("techmate_visits", "1");
    79→    return 1;
    80→  }
    81→  catch { return 1; }
    82→}
    83→function getFirstVisit(): string {
    84→  try {
    85→    const stored = localStorage.getItem("techmate_first_visit");
    86→    if (stored) return stored;
    87→    const now = new Date();
    88→    const formatted = `${now.getDate().toString().padStart(2, "0")}/${(now.getMonth() + 1).toString().padStart(2, "0")}/${now.getFullYear()}`;
    89→    localStorage.setItem("techmate_first_visit", formatted);
    90→    return formatted;
    91→  }
    92→  catch { return "Hoje"; }
    93→}

   193→/* Segmented control for selecting from options */
   194→function SegmentedControl<T extends string>({
   195→  options,
   196→  value,
   197→  onChange,
   198→  renderLabel,
   199→}: {
   200→  options: readonly { label: string; value: T; desc?: string }[];
   201→  value: T;
   202→  onChange: (val: T) => void;
   203→  renderLabel?: (opt: { label: string; value: T; desc?: string }) => string;
   204→}) {
   205→  return (
   206→    <div className="flex flex-wrap gap-1.5">
   207→      {options.map((opt) => {
   208→        const isActive = value === opt.value;
   209→        return (
   210→          <button
   211→            key={opt.value}
   212→            type="button"
   213→            onClick={() => onChange(opt.value)}
   214→            className={`px-3 py-1.5 rounded-lg text-xs font-medium transition-all duration-200 cursor-pointer
   215→                       ${isActive
   216→                         ? "bg-emerald-500/20 border border-emerald-500/30 text-emerald-300 shadow-[0_0_12px_rgba(52,211,153,0.15)]"
   217→                         : "bg-white/[0.03] border border-white/30 text-white/40 hover:bg-white/[0.06] hover:text-white/60"}`}
   218→          >
   219→            {renderLabel ? renderLabel(opt) : opt.label}
   220→            {isActive && <Check className="w-3 h-3 inline ml-1" />}
   221→          </button>
   222→        );
   223→      })}
   224→    </div>
   225→  );
   226→}
